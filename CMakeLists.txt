cmake_minimum_required(VERSION 3.16)
project(SpecQRReader VERSION 0.1.1 LANGUAGES CXX)
option(SPECQR_BUILD_TOOLS "Build standalone PNM/grid command line adapter" ON)
option(SPECQR_BUILD_EXAMPLES "Build public API example" ON)
option(SPECQR_BUILD_TESTS "Build standalone offline tests" OFF)
add_library(specqr_reader src/grid.cpp src/rs.cpp src/payload.cpp src/image.cpp src/common.cpp)
target_compile_features(specqr_reader PUBLIC cxx_std_17)
target_include_directories(specqr_reader PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(SPECQR_BUILD_TOOLS)
  add_executable(specqr-reader tools/reader_cli.cpp)
  target_link_libraries(specqr-reader PRIVATE specqr_reader)
endif()
if(SPECQR_BUILD_EXAMPLES)
  add_executable(specqr-consumer examples/consumer.cpp)
  target_link_libraries(specqr-consumer PRIVATE specqr_reader)
endif()
if(SPECQR_BUILD_TESTS)
  enable_testing()
  foreach(name api_tests payload_rs_tests fuzz_smoke image_boundary_tests fnc1_public_tests)
    add_executable(${name} tests/${name}.cpp)
    target_include_directories(${name} PRIVATE src)
    target_link_libraries(${name} PRIVATE specqr_reader)
    if(MSVC)
      target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
  add_executable(grid_padding_tests tests/grid_padding_tests.cpp)
  target_link_libraries(grid_padding_tests PRIVATE specqr_reader)
  if(MSVC)
    target_compile_options(grid_padding_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(grid_padding_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME grid_padding_tests COMMAND grid_padding_tests ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/grids/segno-v01-H-m0.grid)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/SpecQRReaderConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SpecQRReaderConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpecQRReader)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SpecQRReaderConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS specqr_reader EXPORT SpecQRReaderTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SpecQRReaderTargets NAMESPACE SpecQR:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpecQRReader)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SpecQRReaderConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/SpecQRReaderConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SpecQRReader)
